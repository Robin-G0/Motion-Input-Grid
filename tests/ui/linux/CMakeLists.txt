find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
foreach(application configurator controller)
    if(application STREQUAL "configurator")
        set(app_target motion-input-grid-studio)
        set(run_target run-motion-input-grid-studio)
    else()
        set(app_target motion-input-grid-controller)
        set(run_target run-motion-input-grid-controller)
    endif()
    if(TARGET ${app_target})
        target_sources(${app_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/${application}_ui_tests.cpp")
        target_include_directories(${app_target} PRIVATE "${PROJECT_SOURCE_DIR}/src/linux-apps")
        target_compile_definitions(${app_target} PRIVATE MIG_APP_TESTS)
        add_test(NAME linux-${application}-ui COMMAND ${app_target} --ui-test)
        set_tests_properties(linux-${application}-ui PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30 RUN_SERIAL TRUE)
    endif()
endforeach()
add_executable(mig-qt-theme-tests qt_theme_tests.cpp "${PROJECT_SOURCE_DIR}/src/linux-apps/theme.cpp")
target_link_libraries(mig-qt-theme-tests PRIVATE Qt6::Widgets)
add_test(NAME qt-theme COMMAND mig-qt-theme-tests)
set_tests_properties(qt-theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
